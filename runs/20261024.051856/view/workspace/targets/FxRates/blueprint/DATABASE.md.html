Programmatic Acceptance

=== AC rates-get ===
Intent: A rate can be retrieved by its unique publication date and currency key.

from persistence.testing import db_session

with db_session() as db: created = db.rates.upsert(date="2099-01-02", currency="ZZZ", rate="1.234") assert created is not None row = db.rates.get(date="2099-01-02", currency="ZZZ") assert row is not None assert row.date == created.date assert row.currency == created.currency assert row.rate == created.rate === END AC rates-get ===

=== AC rates-list ===
Intent: Rate rows can be listed and filtered by declared equality fields.

from persistence.testing import db_session

with db_session() as db: created = db.rates.upsert(date="2099-01-03", currency="ZZY", rate="2.345") assert created is not None rows = db.rates.list(date="2099-01-03") assert any(row.date == created.date and row.currency == created.currency and row.rate == created.rate for row in rows) === END AC rates-list ===

=== AC rates-upsert ===
Intent: Storing a rate inserts it and replaces the value for the same date and currency.

from persistence.testing import db_session

with db_session() as db: first = db.rates.upsert(date="2099-01-04", currency="ZZX", rate="3.456") assert first is not None replaced = db.rates.upsert(date="2099-01-04", currency="ZZX", rate="4.567") assert replaced is not None row = db.rates.get(date="2099-01-04", currency="ZZX") assert row is not None assert row.rate == replaced.rate assert row.date == first.date === END AC rates-upsert ===

=== AC rates-delete ===
Intent: A stored rate can be deleted by its unique publication date and currency key.

from persistence.testing import db_session

with db_session() as db: created = db.rates.upsert(date="2099-01-05", currency="ZZW", rate="5.678") assert created is not None deleted = db.rates.delete(date="2099-01-05", currency="ZZW") assert deleted is not None row = db.rates.get(date=created.date, currency=created.currency) assert row is None === END AC rates-delete ===
