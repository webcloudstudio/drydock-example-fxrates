ImplementationGenerated from this file at the DATABASE build step; application code does not open a database driver directly.

The configured database location resolves from DATABASE_URL, then DATABASE_PATH, then data/app.db. Rates are stored as their ECB-published text. Arithmetic is performed by application services with Python decimal.

Programmatic Acceptance

=== AC rates-list-all ===
Intent: Listing rates returns rows with the declared fields through the generated persistence interface.

from persistence.testing import db_session

with db_session() as db: date = "2098-12-30" currency = "QZX" rate_text = "8.765" db.rates.replace_rate(date=date, currency=currency, rate=rate_text) rows = db.rates.list_all() assert any(row.date == date and row.currency == currency and row.rate == rate_text for row in rows) === END AC rates-list-all ===

=== AC rates-latest-on-or-before ===
Intent: The date-bounded listing returns the most recent matching publication first.

from persistence.testing import db_session

with db_session() as db: date = "2098-12-30" currency = "QZY" rate_text = "7.654" db.rates.replace_rate(date=date, currency=currency, rate=rate_text) rows = db.rates.latest_on_or_before(cutoff_date="2098-12-31") assert rows assert rows[0].date <= "2098-12-31" assert any(row.date == date and row.currency == currency and row.rate == rate_text for row in rows) === END AC rates-latest-on-or-before ===

=== AC rates-for-date ===
Intent: A composite date and currency lookup reads back the stored published text.

from persistence.testing import db_session

with db_session() as db: date = "2098-12-29" currency = "QZW" rate_text = "6.543" db.rates.replace_rate(date=date, currency=currency, rate=rate_text) row = db.rates.for_date(date=date, currency=currency) assert row is not None assert row.date == date assert row.currency == currency assert row.rate == rate_text === END AC rates-for-date ===

=== AC rates-for-currency ===
Intent: Currency listing returns the inserted rate in ascending publication-date order.

from persistence.testing import db_session

with db_session() as db: date = "2098-12-28" currency = "QZV" rate_text = "5.432" db.rates.replace_rate(date=date, currency=currency, rate=rate_text) rows = db.rates.for_currency(currency=currency) assert any(row.date == date and row.currency == currency and row.rate == rate_text for row in rows) dates = [row.date for row in rows] assert dates == sorted(dates) === END AC rates-for-currency ===

=== AC rates-history ===
Intent: Inclusive currency history returns the inserted point in ascending date order.

from persistence.testing import db_session

with db_session() as db: date = "2098-12-27" currency = "QZU" rate_text = "4.321" db.rates.replace_rate(date=date, currency=currency, rate=rate_text) rows = db.rates.history(currency=currency, start_date=date, end_date=date) assert len(rows) == 1 assert rows[0].date == date assert rows[0].currency == currency assert rows[0].rate == rate_text === END AC rates-history ===

=== AC rates-replace-rate ===
Intent: Upserting the same date and currency replaces the stored rate text.

from persistence.testing import db_session

with db_session() as db: date = "2098-12-26" currency = "QZT" first_rate = "3.210" replacement_rate = "3.211" db.rates.replace_rate(date=date, currency=currency, rate=first_rate) db.rates.replace_rate(date=date, currency=currency, rate=replacement_rate) row = db.rates.for_date(date=date, currency=currency) assert row is not None assert row.rate == replacement_rate === END AC rates-replace-rate ===

=== AC rates-delete ===
Intent: Deleting a rate removes the matching row from subsequent reads.

from persistence.testing import db_session

with db_session() as db: date = "2098-12-25" currency = "QZS" rate_text = "2.109" db.rates.replace_rate(date=date, currency=currency, rate=rate_text) deleted = db.rates.delete(date=date, currency=currency) assert deleted is not None assert deleted.date == date assert deleted.currency == currency assert db.rates.for_date(date=date, currency=currency) is None === END AC rates-delete ===
