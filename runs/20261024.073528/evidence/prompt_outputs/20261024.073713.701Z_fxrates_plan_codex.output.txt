## Programmatic Acceptance

=== AC rates-list ===
Intent: The rates list operation returns rows matching its optional equality filters.

from persistence.testing import db_session

with db_session() as db:
    date = "2099-01-02"
    currency = "ZZZ"
    rate = "7.031"
    db.rates.upsert(date=date, currency=currency, rate=rate)
    rows = db.rates.list(date=date, currency=currency)
    assert any(row.date == date and row.currency == currency and row.rate == rate for row in rows)
=== END AC rates-list ===

=== AC rates-upsert ===
Intent: The rates upsert operation stores a published rate and replaces the row with the same date and currency.

from persistence.testing import db_session

with db_session() as db:
    date = "2099-01-03"
    currency = "YYY"
    original_rate = "1.250"
    replacement_rate = "1.375"
    db.rates.upsert(date=date, currency=currency, rate=original_rate)
    db.rates.upsert(date=date, currency=currency, rate=replacement_rate)
    rows = db.rates.list(date=date, currency=currency)
    assert len(rows) == 1
    assert rows[0].rate == replacement_rate
=== END AC rates-upsert ===
