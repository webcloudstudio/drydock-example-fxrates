=== AC verification-persistence-boundary ===
Intent: No application module outside the shipped persistence package imports a database driver, so all access goes through the persistence library.
Requires: executable=python3; scope=test

import ast
import os

skip = {".venv", "venv", "persistence", "tests", "sources", ".git", "__pycache__", "node_modules", "blueprint"}
drivers = {"sqlite3", "psycopg", "psycopg2", "sqlalchemy"}
offenders = []
scanned = 0

for root, dirs, files in os.walk("."):
    dirs[:] = [d for d in dirs if d not in skip]
    for name in files:
        if not name.endswith(".py"):
            continue
        path = os.path.join(root, name)
        with open(path, encoding="utf-8") as handle:
            tree = ast.parse(handle.read(), filename=path)
        scanned += 1
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                mods = [alias.name for alias in node.names]
            elif isinstance(node, ast.ImportFrom):
                mods = [node.module or ""]
            else:
                continue
            for mod in mods:
                if mod.split(".")[0] in drivers:
                    offenders.append((path, mod))

print(f"scanned {scanned} modules, {len(offenders)} driver imports")
assert scanned > 0
assert offenders == []
=== END AC verification-persistence-boundary ===
