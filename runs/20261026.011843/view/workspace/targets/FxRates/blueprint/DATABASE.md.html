Programmatic Acceptance

=== AC rates-upsert ===
Intent: The rates upsert operation replaces an existing date and currency rate and returns the stored row.

from persistence.testing import db_session

date = "2099-01-02" currency = "ZZZ" original_rate = "1.25" replacement_rate = "1.2500"

with db_session() as db: db.rates.upsert(date=date, currency=currency, rate=original_rate) stored = db.rates.upsert(date=date, currency=currency, rate=replacement_rate) assert (stored.date, stored.currency, stored.rate) == (date, currency, replacement_rate) === END AC rates-upsert ===

=== AC rates-effective-rates ===
Intent: Effective-date lookup returns stored rows on or before its bound in descending date order.

from persistence.testing import db_session

first_date = "2099-02-01" second_date = "2099-02-03" currency = "ZZZ" rate = "2.5"

with db_session() as db: db.rates.upsert(date=first_date, currency=currency, rate=rate) db.rates.upsert(date=second_date, currency=currency, rate=rate) rows = db.rates.effective_rates(as_of=second_date) matching = [row.date for row in rows if row.currency == currency and row.date in (first_date, second_date)] assert matching == sorted((first_date, second_date), reverse=True) === END AC rates-effective-rates ===

=== AC rates-by-date ===
Intent: Date lookup returns the stored rows for the requested date.

from persistence.testing import db_session

date = "2099-03-01" currency = "ZZZ" rate = "3.75"

with db_session() as db: db.rates.upsert(date=date, currency=currency, rate=rate) rows = db.rates.by_date(date=date) matching = [(row.date, row.currency, row.rate) for row in rows if row.date == date and row.currency == currency] assert matching == [(date, currency, rate)] === END AC rates-by-date ===

=== AC rates-history ===
Intent: Currency history returns matching rows in ascending date order within its inclusive bounds.

from persistence.testing import db_session

currency = "ZZZ" first_date = "2099-04-01" second_date = "2099-04-02" outside_date = "2099-04-03" rate = "4.5"

with db_session() as db: db.rates.upsert(date=first_date, currency=currency, rate=rate) db.rates.upsert(date=second_date, currency=currency, rate=rate) db.rates.upsert(date=outside_date, currency=currency, rate=rate) rows = db.rates.history(currency=currency, start_date=first_date, end_date=second_date) matching = [(row.date, row.currency, row.rate) for row in rows if row.currency == currency] assert matching == [(first_date, currency, rate), (second_date, currency, rate)] === END AC rates-history ===

=== AC rates-list ===
Intent: The rates list operation reads back all rows created by the criterion in declared date and currency order.

from persistence.testing import db_session

first_date = "2099-05-01" second_date = "2099-05-02" first_currency = "AAA" second_currency = "ZZZ" rate = "5.5"

with db_session() as db: db.rates.upsert(date=second_date, currency=first_currency, rate=rate) db.rates.upsert(date=first_date, currency=second_currency, rate=rate) db.rates.upsert(date=first_date, currency=first_currency, rate=rate) rows = db.rates.list() matching = [(row.date, row.currency, row.rate) for row in rows if row.date in (first_date, second_date) and row.currency in (first_currency, second_currency)] assert matching == [ (first_date, first_currency, rate), (first_date, second_currency, rate), (second_date, first_currency, rate), ] === END AC rates-list ===
