## Programmatic Acceptance

=== AC rates-list-roundtrip ===
Intent: The declared rates.list operation returns rows created through the repository.
from persistence.testing import db_session

with db_session() as db:
    marker_date = "2098-11-17"
    marker_currency = "XQZ"
    marker_rate = "3.1400"
    created = db.rates.replace_rate(
        date=marker_date, currency=marker_currency, rate=marker_rate
    )
    assert created.date == marker_date
    rows = db.rates.list(date=marker_date, currency=marker_currency)
    assert any(
        row.date == marker_date
        and row.currency == marker_currency
        and row.rate == marker_rate
        for row in rows
    )
=== END AC rates-list-roundtrip ===

=== AC rates-replace-roundtrip ===
Intent: Replacing a date and currency returns the newly stored text.
from persistence.testing import db_session

with db_session() as db:
    marker_date = "2098-11-18"
    marker_currency = "XQZ"
    first_rate = "2.500"
    replacement_rate = "2.7500"
    db.rates.replace_rate(date=marker_date, currency=marker_currency, rate=first_rate)
    replaced = db.rates.replace_rate(
        date=marker_date, currency=marker_currency, rate=replacement_rate
    )
    assert replaced.rate == replacement_rate
    stored = db.rates.list(date=marker_date, currency=marker_currency)
    assert len(stored) == 1
    assert stored[0].rate == replacement_rate
=== END AC rates-replace-roundtrip ===
