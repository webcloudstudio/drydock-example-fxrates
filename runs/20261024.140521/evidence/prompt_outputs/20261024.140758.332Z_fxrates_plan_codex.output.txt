## Programmatic Acceptance

=== AC rates-list-readback ===
Intent: Read stored rate rows through the declared list operation.

from persistence.testing import db_session

with db_session() as db:
    rows_before = db.rates.list()
    marker_date = "2099-01-01"
    marker_currency = "ZZZ"
    marker_rate = "9.876"
    created = db.rates.replace_rate(
        date=marker_date, currency=marker_currency, rate=marker_rate
    )
    rows_after = db.rates.list()
    assert len(rows_after) == len(rows_before) + 1
    assert any(
        row.date == marker_date
        and row.currency == marker_currency
        and row.rate == marker_rate
        for row in rows_after
    )

=== END AC rates-list-readback ===

=== AC rates-find-by-date ===
Intent: List every stored currency rate for a publication date.

from persistence.testing import db_session

with db_session() as db:
    marker_date = "2099-01-03"
    first_currency = "ZZX"
    first_rate = "7.654"
    second_currency = "ZZW"
    second_rate = "6.543"
    db.rates.replace_rate(date=marker_date, currency=first_currency, rate=first_rate)
    db.rates.replace_rate(date=marker_date, currency=second_currency, rate=second_rate)
    rows = db.rates.find_by_date(date=marker_date)
    observed = {(row.currency, row.rate) for row in rows}
    assert observed == {(first_currency, first_rate), (second_currency, second_rate)}

=== END AC rates-find-by-date ===

=== AC rates-history-bounds ===
Intent: List a currency's rates within inclusive date bounds in ascending order.

from persistence.testing import db_session

with db_session() as db:
    currency = "ZZU"
    before_date = "2099-01-04"
    start_date = "2099-01-05"
    end_date = "2099-01-06"
    after_date = "2099-01-07"
    before_rate = "2.100"
    first_rate = "2.200"
    second_rate = "2.300"
    after_rate = "2.400"
    for date, rate in (
        (before_date, before_rate),
        (start_date, first_rate),
        (end_date, second_rate),
        (after_date, after_rate),
    ):
        db.rates.replace_rate(date=date, currency=currency, rate=rate)
    rows = db.rates.history(currency=currency, start_date=start_date, end_date=end_date)
    observed = [(row.date, row.rate) for row in rows]
    assert observed == [(start_date, first_rate), (end_date, second_rate)]

## Guardrails

- Rates are stored as published text; do not convert stored values to floating point.
- Natural uniqueness is `(date, currency)`; retain a separate single-column primary key.
- Application code accesses storage only through the generated persistence library.
- Do not assume an empty store in acceptance checks.
