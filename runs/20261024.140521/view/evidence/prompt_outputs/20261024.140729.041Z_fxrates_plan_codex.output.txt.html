=== AC rate-upsert ===
Intent: A rate written through the declared repository operation can be read back with its supplied values.

from persistence.testing import db_session

with db_session() as db:
    date = "2099-12-30"
    currency = "ZZQ"
    rate = "1.23450"
    db.rates.upsert(date=date, currency=currency, rate=rate)
    row = db.rates.get(date=date, currency=currency)
    assert row is not None
    assert (row.date, row.currency, row.rate) == (date, currency, rate)
