Programmatic Acceptance

=== AC database-upsert-rate ===
Intent: A rate upsert persists its input fields and can be read back by its natural key.

from persistence.testing import db_session

date = "2099-01-02" currency = "TST" rate = "1.2300" with db_session() as db: db.rates.upsert(date=date, currency=currency, rate=rate) stored = db.rates.get(date=date, currency=currency) assert stored is not None assert (stored.date, stored.currency, stored.rate) == (date, currency, rate) === END AC database-upsert-rate ===
