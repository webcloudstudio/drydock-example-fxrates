#!/bin/sh
# run.sh — FxRates grader. Do not filter, skip, or reinterpret.
#
# Runs ./fxrates against a fresh database: loads the supplied ECB files from the command line,
# starts `./fxrates serve` on a free local port, and compares every command's exit code and
# standard output, and every HTTP response's status and JSON body, with the known answer below.
# JSON bodies are compared as values, so key order and whitespace do not matter. Standard error
# and error-response bodies are never compared. No network beyond 127.0.0.1 is used.
#
#   sh sources/run.sh               from the application root (how `drydock uat` runs it)
#   sh uat/FxRates/bin/run.sh APP   from anywhere, grading the application root APP
#
# Exit 0 only when every check passes.
set -u

HERE=$(cd "$(dirname "$0")" && pwd)
APP=$(cd "${1:-.}" && pwd)
if [ -f "$HERE/eurofxref-hist_2025-01.csv" ]; then DATA=$HERE; else DATA=$(cd "$HERE/../sources" && pwd); fi
WORK=$(mktemp -d)
BIN="$APP/fxrates"
DATABASE_URL="$WORK/fxrates.db"
export DATABASE_URL
SERVER=""
PORT=$(python3 -c 'import socket; s = socket.socket(); s.bind(("127.0.0.1", 0)); print(s.getsockname()[1])')
BASE="http://127.0.0.1:$PORT"

stop_server() {
    if [ -n "$SERVER" ]; then
        pkill -TERM -P "$SERVER" 2>/dev/null
        kill -TERM "$SERVER" 2>/dev/null
        wait "$SERVER" 2>/dev/null
        SERVER=""
    fi
}
trap 'stop_server; rm -rf "$WORK"' EXIT
trap 'exit 1' INT TERM

if [ ! -x "$BIN" ]; then
    echo "error: no executable ./fxrates at the application root $APP." >&2
    echo "The deliverable is an executable named fxrates at the application root." >&2
    exit 1
fi

PASSED=0
FAILED=0

pass() { PASSED=$((PASSED + 1)); echo "PASS $1"; }
fail() { FAILED=$((FAILED + 1)); echo "FAIL $1"; }

# check <id> <expected exit> <fxrates args...>  — expected standard output on stdin.
check() {
    id=$1
    want_rc=$2
    shift 2
    cat > "$WORK/want"
    (cd "$APP" && "$BIN" "$@") > "$WORK/got" 2> "$WORK/err" < /dev/null
    rc=$?
    if [ "$rc" -eq "$want_rc" ] && cmp -s "$WORK/want" "$WORK/got"; then
        pass "$id: fxrates $*"
    else
        fail "$id: fxrates $*"
        echo "  exit: expected $want_rc, got $rc"
        diff "$WORK/want" "$WORK/got" | sed 's/^/  /'
        sed 's/^/  stderr: /' "$WORK/err"
    fi
}

# http <id> <expected status> <path>  — expected JSON body on stdin (compared only for 200).
http() {
    id=$1
    want_status=$2
    path=$3
    cat > "$WORK/want"
    if python3 - "$BASE$path" "$want_status" "$WORK/want" > "$WORK/got" 2>&1 <<'PY'
import json, sys, urllib.error, urllib.request
url, want, want_file = sys.argv[1], int(sys.argv[2]), sys.argv[3]
try:
    with urllib.request.urlopen(url, timeout=15) as r:
        status, body = r.status, r.read()
except urllib.error.HTTPError as e:
    status, body = e.code, e.read()
except Exception as e:
    print(f"request failed: {e}")
    sys.exit(1)
if status != want:
    print(f"status: expected {want}, got {status}")
    print(body.decode("utf-8", "replace")[:400])
    sys.exit(1)
if want != 200:
    sys.exit(0)
try:
    got = json.loads(body)
except ValueError:
    print("body is not JSON:", body.decode("utf-8", "replace")[:400])
    sys.exit(1)
expected = json.loads(open(want_file, encoding="utf-8").read())
if got != expected:
    print("expected:", json.dumps(expected, sort_keys=True))
    print("got:     ", json.dumps(got, sort_keys=True))
    sys.exit(1)
PY
    then
        pass "$id: GET $path"
    else
        fail "$id: GET $path"
        sed 's/^/  /' "$WORK/got"
    fi
}

# start_server — run `fxrates serve` in the background and wait up to 30 s for /health.
start_server() {
    (cd "$APP" && exec "$BIN" serve --port "$PORT") > "$WORK/server.log" 2>&1 < /dev/null &
    SERVER=$!
    tries=0
    until python3 -c "import urllib.request, sys; urllib.request.urlopen(sys.argv[1], timeout=2)" \
            "$BASE/health" 2>/dev/null; do
        tries=$((tries + 1))
        if [ "$tries" -ge 30 ] || ! kill -0 "$SERVER" 2>/dev/null; then
            echo "FAIL S1: fxrates serve --port $PORT did not answer GET /health within 30 s"
            sed 's/^/  server: /' "$WORK/server.log"
            echo "FxRates: $PASSED passed, unscored HTTP checks (server did not start)"
            exit 1
        fi
        sleep 1
    done
}
