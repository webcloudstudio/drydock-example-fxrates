Programmatic Acceptance

=== AC rates-replace-rate ===
Intent: Storing a rate for an existing (date, currency) replaces its text rather than adding a row, and the read-back returns the second text exactly.
Requires: executable=python3; scope=test

from persistence.testing import db_session

d = "1900-01-02" c = "ZZA" first = "1.50" second = "2.5"

with db_session() as db: db.rates.replace_rate(date=d, currency=c, rate=first) row = db.rates.replace_rate(date=d, currency=c, rate=second) assert row.rate == second rows = [r for r in db.rates.list_for_date(date=d) if r.currency == c] assert len(rows) == 1 assert rows[0].rate == second === END AC rates-replace-rate ===

=== AC rates-list-for-date ===
Intent: Listing one date returns only that date's rates, ordered by currency ascending.
Requires: executable=python3; scope=test

from persistence.testing import db_session

d = "1900-02-01" other = "1900-02-02" codes = ["ZZC", "ZZA", "ZZB"] stray = "ZZD"

with db_session() as db: for i, code in enumerate(codes): db.rates.replace_rate(date=d, currency=code, rate=str(i + 1)) db.rates.replace_rate(date=other, currency=stray, rate="9") rows = db.rates.list_for_date(date=d) got = [r.currency for r in rows if r.currency in codes or r.currency == stray] assert got == sorted(codes) assert all(r.date == d for r in rows) === END AC rates-list-for-date ===

=== AC rates-list-on-or-before ===
Intent: Dates come back latest first and bounded by as_of inclusive, so the first row is the effective date for a request between two publications.
Requires: executable=python3; scope=test

from persistence.testing import db_session

dates = ["1900-03-01", "1900-03-05", "1900-03-09"] asof = "1900-03-07" c = "ZZA"

with db_session() as db: for d in dates: db.rates.replace_rate(date=d, currency=c, rate="1") rows = db.rates.list_on_or_before(as_of=asof) mine = [r.date for r in rows if r.date in dates] assert mine == sorted([d for d in dates if d <= asof], reverse=True) assert rows[0].date == max(d for d in dates if d <= asof) exact = db.rates.list_on_or_before(as_of=dates[1]) assert exact[0].date == dates[1] early = db.rates.list_on_or_before(as_of="1900-01-01") assert [r.date for r in early if r.date in dates] == [] === END AC rates-list-on-or-before ===

=== AC rates-list-all-keys ===
Intent: Every stored key at or after the lower bound is listed in ascending date order, and a bound after a row excludes it.
Requires: executable=python3; scope=test

from persistence.testing import db_session

dates = ["1900-04-03", "1900-04-01", "1900-04-02"] codes = ["ZZA", "ZZB"] since = "1900-04-02"

with db_session() as db: for d in dates: for code in codes: db.rates.replace_rate(date=d, currency=code, rate="3") rows = db.rates.list_all_keys(since=since) mine = [(r.date, r.currency) for r in rows if r.currency in codes and r.date.startswith("1900-04")] expected = sorted((d, code) for d in dates for code in codes if d >= since) assert mine == expected === END AC rates-list-all-keys ===

=== AC rates-list-currency-history ===
Intent: A currency's history is filtered by inclusive optional bounds and sorted by date ascending; an inverted range is empty.
Requires: executable=python3; scope=test

from persistence.testing import db_session

dates = ["1900-06-04", "1900-06-01", "1900-06-03", "1900-06-02"] c = "ZZA" other = "ZZB" start = "1900-06-02" end = "1900-06-03"

with db_session() as db: for d in dates: db.rates.replace_rate(date=d, currency=c, rate="5") db.rates.replace_rate(date=dates[0], currency=other, rate="6") bounded = db.rates.list_currency_history(currency=c, start_date=start, end_date=end) assert [r.date for r in bounded] == sorted(d for d in dates if start <= d <= end) everything = db.rates.list_currency_history(currency=c) assert [r.date for r in everything] == sorted(dates) assert all(r.currency == c for r in everything) inverted = db.rates.list_currency_history(currency=c, start_date=end, end_date=start) assert inverted == [] === END AC rates-list-currency-history ===

=== AC rates-delete-from-date ===
Intent: Deleting from a date removes rates on or after it and leaves earlier ones, and reports how many rows went.
Requires: executable=python3; scope=test

from persistence.testing import db_session

keep = "1900-07-01" drop = "1900-07-02" c = "ZZA"

with db_session() as db: db.rates.replace_rate(date=keep, currency=c, rate="7") db.rates.replace_rate(date=drop, currency=c, rate="8") removed = db.rates.delete_from_date(from_date=drop) assert removed >= 1 remaining = db.rates.list_currency_history(currency=c) assert [r.date for r in remaining] == [keep] === END AC rates-delete-from-date ===
