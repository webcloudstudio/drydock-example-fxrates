Programmatic Acceptance

=== AC rates-upsert ===
Intent: Upserting a rate stores its published text and replaces the matching date and currency row.

from persistence.testing import db_session

date = "2099-01-02" currency = "ZZZ" published = "0.943" replacement = "0.944" with db_session() as db: db.rates.upsert(date=date, currency=currency, rate=published) first = db.rates.get(date=date, currency=currency) assert first.rate == published db.rates.upsert(date=date, currency=currency, rate=replacement) stored = db.rates.get(date=date, currency=currency) assert stored.rate == replacement === END AC rates-upsert ===

=== AC rates-list-date ===
Intent: Listing rates by date returns the rate rows stored for that date.

from persistence.testing import db_session

date = "2099-01-03" currency = "ZZZ" published = "1.25" with db_session() as db: db.rates.upsert(date=date, currency=currency, rate=published) rows = db.rates.list_date(date=date) assert any(row.date == date and row.currency == currency and row.rate == published for row in rows) === END AC rates-list-date ===
