## Programmatic Acceptance

=== AC publication-dates ===
Intent: The repository lists stored publication dates and supports statistics derived from those dates.

from persistence.testing import db_session

with db_session() as db:
    suffix = "db-ac-publication"
    first = db.rates.replace_rate(
        date="2098-01-01", currency="USD", rate="1.25"
    )
    second = db.rates.replace_rate(
        date="2098-01-02", currency="GBP", rate="0.75"
    )
    dates = db.rates.publication_stats()
    assert first.date == "2098-01-01"
    assert second.date == "2098-01-02"
    assert [row.date for row in dates if row.date.startswith("2098-01-")] == [
        "2098-01-01", "2098-01-02"
    ]
=== END AC ===

=== AC rates-for-date ===
Intent: A date lookup returns the stored rate rows for that date in currency order.

from persistence.testing import db_session

with db_session() as db:
    db.rates.replace_rate(date="2098-02-01", currency="USD", rate="1.25")
    db.rates.replace_rate(date="2098-02-01", currency="GBP", rate="0.75")
    rows = db.rates.for_date(date="2098-02-01")
    assert [(row.currency, row.rate) for row in rows] == [
        ("GBP", "0.75"), ("USD", "1.25")
    ]
=== END AC ===

=== AC currencies ===
Intent: Currency listing returns stored currencies in ascending order.

from persistence.testing import db_session

with db_session() as db:
    db.rates.replace_rate(date="2098-03-01", currency="USD", rate="1.25")
    db.rates.replace_rate(date="2098-03-01", currency="GBP", rate="0.75")
    rows = db.rates.currencies()
    selected = [row.currency for row in rows if row.currency in {"USD", "GBP"}]
    assert selected == ["GBP", "USD"]
=== END AC ===

=== AC rate-get ===
Intent: A stored rate can be retrieved by its date and currency.

from persistence.testing import db_session

with db_session() as db:

    row = db.rates.get(date="2098-04-01", currency="USD")

    assert (row.date, row.currency, row.rate) == ("2098-04-01", "USD", "1.25")

=== AC rate-history ===
Intent: History returns the selected currency's rows in ascending date order.

from persistence.testing import db_session

with db_session() as db:
    db.rates.replace_rate(date="2098-05-02", currency="USD", rate="1.25")
    db.rates.replace_rate(date="2098-05-01", currency="USD", rate="1.20")
    rows = db.rates.history(
        currency="USD", start_date="2098-05-01", end_date="2098-05-02"
    )
    assert [(row.date, row.rate) for row in rows] == [
        ("2098-05-01", "1.20"), ("2098-05-02", "1.25")
    ]
=== END AC ===

=== AC replace-rate ===
Intent: Replacing a date and currency stores the supplied rate text exactly and replaces the prior value.

from persistence.testing import db_session

with db_session() as db:

=== AC clear-rates ===
Intent: The clear operation removes only the criterion's selected date and currency rows.

from persistence.testing import db_session

with db_session() as db:
    db.rates.replace_rate(date="2098-07-01", currency="USD", rate="1.25")
    db.rates.replace_rate(date="2098-07-01", currency="GBP", rate="0.75")
    db.rates.replace_rate(date="2098-07-02", currency="USD", rate="1.30")
    db.rates.delete(date="2098-07-01", currency="USD")
    removed = db.rates.get(date="2098-07-01", currency="USD")
    preserved = db.rates.get(date="2098-07-01", currency="GBP")
    assert removed is None
    assert preserved is not None
    assert preserved.rate == "0.75"
=== END AC ===

## Guardrails

- Store rates as the ECB's published text; do not convert stored rates to floating point.
- Keep database access behind the generated persistence library.
- Resolve the database from DATABASE_URL, then DATABASE_PATH, then data/app.db.
- Do not read the supplied grader or data files from application code.
